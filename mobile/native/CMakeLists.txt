cmake_minimum_required(VERSION 3.22)
project(cinema LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(CURL REQUIRED)
find_package(LibXml2 REQUIRED)
find_package(json-c CONFIG REQUIRED)

# Certyfikaty są częścią biblioteki, więc nie zależą od katalogu roboczego telefonu.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/cacert.pem" CERTIFICATES)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/certificates.h" "static const char certificates[] = R\"CERT(${CERTIFICATES})CERT\";\n")
if(ANDROID)
    add_library(cinema SHARED engine.cpp)
    target_link_options(cinema PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_library(cinema STATIC engine.cpp)
endif()
target_include_directories(cinema PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(cinema PRIVATE CURL::libcurl LibXml2::LibXml2 json-c::json-c)
if(DEFINED VCPKG_INSTALLED_DIR)
    # Licencje bibliotek statycznych trafiają razem z aplikacją na telefon.
    file(GLOB LICENSE_FILES "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/*/copyright")
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/licenses")
    foreach(LICENSE_FILE IN LISTS LICENSE_FILES)
        get_filename_component(PORT_DIR "${LICENSE_FILE}" DIRECTORY)
        get_filename_component(PORT "${PORT_DIR}" NAME)
        configure_file("${LICENSE_FILE}" "${CMAKE_BINARY_DIR}/licenses/${PORT}.txt" COPYONLY)
    endforeach()
endif()
if(NOT ANDROID)
    enable_testing()
    add_executable(mobile-tests ../../tests/mobile-native.cpp)
    target_link_libraries(mobile-tests PRIVATE cinema)
    add_test(NAME mobile-bridge COMMAND mobile-tests)
endif()
